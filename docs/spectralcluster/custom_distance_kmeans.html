 13def run_kmeans(spectral_embeddings: np.ndarray,
 14               n_clusters: int,
 15               custom_dist: typing.Union[str, typing.Callable],
 16               max_iter: int,
 17               random_state: int = 0) -> np.ndarray:
 18  """Run CustomKMeans with a custom distance measure support.
 19
 20  Perform a custom kmeans clustering with any distance measure defined
 21  in scipy.spatial.distance.
 22
 23  Args:
 24    spectral_embeddings: input spectral embedding observations
 25    n_clusters: the number of clusters to form
 26    custom_dist: str or callable. custom distance measure for k-means. if a
 27      string, "cosine", "euclidean", "mahalanobis", or any other distance
 28      functions defined in scipy.spatial.distance can be used
 29    max_iter: the maximum number of iterations for the custom k-means
 30    random_state: random seed for k-means initialization
 31
 32  Returns:
 33    labels: predicted clustering labels of all samples
 34  """
 35  if not custom_dist:  # Scikit-learn KMeans
 36    kmeans_clusterer = KMeans(
 37        n_clusters=n_clusters, init="k-means++", max_iter=max_iter,
 38        random_state=random_state, n_init="auto")
 39    return kmeans_clusterer.fit_predict(spectral_embeddings)
 40  else:
 41    # Initialization using the k-means++ method in Scikit-learn
 42    kmeans_clusterer = KMeans(
 43        n_clusters=n_clusters, init="k-means++", max_iter=1,
 44        random_state=random_state, n_init="auto")
 45    kmeans_clusterer.fit(spectral_embeddings)
 46    centroids = kmeans_clusterer.cluster_centers_
 47    # Run the custom K-means
 48    kmeans_clusterer = CustomKMeans(
 49        n_clusters=n_clusters,
 50        centroids=centroids,
 51        max_iter=max_iter,
 52        custom_dist=custom_dist,
 53        random_state=random_state)
 54
 55  labels = kmeans_clusterer.predict(spectral_embeddings)
 56  return labels

 82  random_state: int = 0

 84  def _init_centroids(self, embeddings: np.ndarray):
 85    """Compute the initial centroids."""
 86
 87    n_samples = embeddings.shape[0]
 88    rng = np.random.RandomState(self.random_state)
 89    idx = rng.choice(
 90        np.arange(n_samples), size=self.n_clusters, replace=False)
 91    self.centroids = embeddings[idx, :]
 92
 93  def predict(self, embeddings: np.ndarray) -> np.ndarray:
 94    """Performs the clustering.
 95
 96    Args:
 97      embeddings: the input observations to cluster
 98
 99    Returns:
100      labels: predicted clustering labels of all samples
101
102    Raises:
103      ValueError: if input observations have wrong shape
104    """
105    n_samples, n_features = embeddings.shape
106    if self.max_iter <= 0:
107      raise ValueError("Number of iterations should be a positive number,"
108                       " got %d instead" % self.max_iter)
109    if n_samples < self.n_clusters:
110      raise ValueError("n_samples=%d should be >= n_clusters=%d" %
111                       (n_samples, self.n_clusters))
112    if self.centroids is None:
113      self._init_centroids(embeddings)
114    n_centroids, c_n_features = self.centroids.shape
115    if n_centroids != self.n_clusters:
116      raise ValueError("The shape of the initial centroids (%s) "
117                       "does not match the number of clusters %d" %
118                       (str(self.centroids.shape), self.n_clusters))
119    if n_features != c_n_features:
120      raise ValueError(
121          "The number of features of the initial centroids %d "
122          "does not match the number of features of the data %d." %
123          (c_n_features, n_features))
124
125    sample_ids = np.arange(n_samples)
126    prev_mean_dist = 0
127    for iter_idx in range(self.max_iter + 1):
128      # Compute distances to all centroids and assign each sample a label
129      # corresponding to the nearest centroid
130      dist_to_all_centroids = distance.cdist(
131          embeddings, self.centroids, metric=self.custom_dist)
132      labels = dist_to_all_centroids.argmin(axis=1)
133      distances = dist_to_all_centroids[sample_ids, labels]
134      mean_distance = np.mean(distances)
135      # If the difference between current mean_distance and previous
136      # mean_distance is very small or the max iteration number is reached,
137      # the clustering stops
138      if mean_distance <= prev_mean_dist and mean_distance >= (
139          1 - self.tol) * prev_mean_dist or iter_idx == self.max_iter:
140        break
141      prev_mean_dist = mean_distance
142      # Update centroids
143      for each_centroid_idx in range(n_centroids):
144        each_centroid_samples = np.where(labels == each_centroid_idx)[0]
145        if each_centroid_samples.any():
146          self.centroids[each_centroid_idx] = np.mean(
147              embeddings[each_centroid_samples], axis=0)
148    return labels
