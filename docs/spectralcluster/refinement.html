 74class RefinementOptions:
 75  """Refinement options for the affinity matrix."""
 76
 77  # Sigma value of the Gaussian blur operation.
 78  gaussian_blur_sigma: int = 1
 79
 80  # The p-percentile for the row wise thresholding.
 81  p_percentile: float = 0.95
 82
 83  # The multiplier for soft threshold, if this value is 0,
 84  # then it's a hard thresholding.
 85  thresholding_soft_multiplier: float = 0.01
 86
 87  # The type of thresholding operation.
 88  thresholding_type: ThresholdType = ThresholdType.RowMax
 89
 90  # If true, we set values larger than the threshold to 1.

 92
 93  # If true, in the row wise thresholding operation, we firstly set diagonals
 94  # of the affinity matrix to 0, and set the diagonals back to 1 in the end.
 95  thresholding_preserve_diagonal: bool = False
 96
 97  # A SymmetrizeType.
 98  symmetrize_type: SymmetrizeType = SymmetrizeType.Max
 99
100  # A list of RefinementName for the sequence of refinement operations to
101  # apply on the affinity matrix.
102  refinement_sequence: typing.Optional[typing.Sequence[RefinementName]] = None

104  def get_refinement_operator(self, name: RefinementName) -> (
105      AffinityRefinementOperation):
106    """Get the refinement operator for the affinity matrix.
107
108    Args:
109      name: a RefinementName
110
111    Returns:
112      object of the operator
113
114    Raises:
115      TypeError: if name is not a RefinementName
116      ValueError: if name is an unknown refinement operation
117    """
118    if name == RefinementName.CropDiagonal:
119      return CropDiagonal()
120    elif name == RefinementName.GaussianBlur:
121      return GaussianBlur(self.gaussian_blur_sigma)
122    elif name == RefinementName.RowWiseThreshold:
123      return RowWiseThreshold(self.p_percentile,
124                              self.thresholding_soft_multiplier,
125                              self.thresholding_type,
126                              self.thresholding_with_binarization,
127                              self.thresholding_preserve_diagonal)
128    elif name == RefinementName.Symmetrize:
129      return Symmetrize(self.symmetrize_type)
130    elif name == RefinementName.Diffuse:
131      return Diffuse()
132    elif name == RefinementName.RowWiseNormalize:
133      return RowWiseNormalize()
134    else:
135      raise ValueError("Unknown refinement operation: {}".format(name))

138class CropDiagonal(AffinityRefinementOperation):
139  """Crop the diagonal.
140
141  Replace diagonal element by the max non-diagonal value of row.
142  After this operation, the matrix has similar properties to a standard
143  Laplacian matrix. This also helps to avoid the bias during Gaussian blur and
144  normalization.
145  """
146
147  def refine(self, affinity: np.ndarray) -> np.ndarray:
148    self.check_input(affinity)
149    refined_affinity = np.copy(affinity)
150    np.fill_diagonal(refined_affinity, 0.0)
151    di = np.diag_indices(refined_affinity.shape[0])
152    refined_affinity[di] = refined_affinity.max(axis=1)
153    return refined_affinity

167class RowWiseThreshold(AffinityRefinementOperation):
168  """Apply row wise thresholding."""
169
170  def __init__(self,
171               p_percentile: float = 0.95,
172               thresholding_soft_multiplier: float = 0.01,
173               thresholding_type: ThresholdType = ThresholdType.RowMax,
174               thresholding_with_binarization: bool = False,
175               thresholding_preserve_diagonal: bool = False):
176    self.p_percentile = p_percentile
177    self.multiplier = thresholding_soft_multiplier
178    if not isinstance(thresholding_type, ThresholdType):
179      raise TypeError("thresholding_type must be a ThresholdType")
180    self.thresholding_type = thresholding_type
181    self.thresholding_with_binarization = thresholding_with_binarization
182    self.thresholding_preserve_diagonal = thresholding_preserve_diagonal
183
184  def refine(self, affinity: np.ndarray) -> np.ndarray:
185    self.check_input(affinity)
186    refined_affinity = np.copy(affinity)
187    if self.thresholding_preserve_diagonal:
188      np.fill_diagonal(refined_affinity, 0.0)
189    if self.thresholding_type == ThresholdType.RowMax:
190      # Row_max based thresholding
191      row_max = refined_affinity.max(axis=1)
192      row_max = np.expand_dims(row_max, axis=1)
193      is_smaller = refined_affinity < (row_max * self.p_percentile)
194    elif self.thresholding_type == ThresholdType.Percentile:
195      # Percentile based thresholding
196      row_percentile = np.percentile(
197          refined_affinity, self.p_percentile * 100, axis=1)
198      row_percentile = np.expand_dims(row_percentile, axis=1)
199      is_smaller = refined_affinity < row_percentile
200    else:
201      raise ValueError("Unsupported thresholding_type")
202    if self.thresholding_with_binarization:
203      # For values larger than the threshold, we binarize them to 1
204      refined_affinity = (np.ones_like(
205          (refined_affinity)) * np.invert(is_smaller)) + (
206              refined_affinity * self.multiplier * is_smaller)
207    else:
208      refined_affinity = (refined_affinity * np.invert(is_smaller)) + (
209          refined_affinity * self.multiplier * is_smaller)
210    if self.thresholding_preserve_diagonal:
211      np.fill_diagonal(refined_affinity, 1.0)
212    return refined_affinity

215class Symmetrize(AffinityRefinementOperation):
216  """The Symmetrization operation."""
217
218  def __init__(self, symmetrize_type: SymmetrizeType = SymmetrizeType.Max):
219    self.symmetrize_type = symmetrize_type
220
221  def refine(self, affinity: np.ndarray) -> np.ndarray:
222    self.check_input(affinity)
223    if self.symmetrize_type == SymmetrizeType.Max:
224      return np.maximum(affinity, np.transpose(affinity))
225    elif self.symmetrize_type == SymmetrizeType.Average:
226      return 0.5 * (affinity + np.transpose(affinity))
227    else:
228      raise ValueError("Unsupported symmetrize_type.")

235    self.check_input(affinity)
236    return np.matmul(affinity, np.transpose(affinity))

239class RowWiseNormalize(AffinityRefinementOperation):
240  """The row wise max normalization operation."""
241
242  def refine(self, affinity: np.ndarray) -> np.ndarray:
243    self.check_input(affinity)
244    refined_affinity = np.copy(affinity).astype(float)
245    row_max = refined_affinity.max(axis=1)
246    refined_affinity /= np.expand_dims(np.maximum(row_max, EPS), axis=1)
247    return refined_affinity

194      is_smaller = refined_affinity < (row_max * self.p_percentile)
195    elif self.thresholding_type == ThresholdType.Percentile:
196      # Percentile based thresholding
197      row_percentile = np.percentile(
198          refined_affinity, self.p_percentile * 100, axis=1)
199      row_percentile = np.expand_dims(row_percentile, axis=1)
200      is_smaller = refined_affinity < row_percentile
201    else:
202      raise ValueError("Unsupported thresholding_type")
203    if self.thresholding_with_binarization:
204      # For values larger than the threshold, we binarize them to 1
205      refined_affinity = (np.ones_like(
206          (refined_affinity)) * np.invert(is_smaller)) + (
207              refined_affinity * self.multiplier * is_smaller)
208    else:
209      refined_affinity = (refined_affinity * np.invert(is_smaller)) + (
210          refined_affinity * self.multiplier * is_smaller)
211    if self.thresholding_preserve_diagonal:
212      np.fill_diagonal(refined_affinity, 1.0)
213    return refined_affinity
